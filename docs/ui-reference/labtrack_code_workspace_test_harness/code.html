Core Objective

Design a data structure that follows the constraints of a Least Recently Used (LRU) Cache with strict O(1) average time complexity for both operations.

#include <unordered_map>
#include <iostream>
// Institutional Laboratory 06: Doubly Linked Hash Cache
using namespace std;
struct Node {
int key, val;
Node* prev;
Node* next;
Node(int k, int v) : key(k), val(v), prev(nullptr), next(nullptr) {}
};
class LRUCache {
private:
unordered_map<int, Node*> cacheMap;
int capacity;
Node* head;
Node* tail;
void removeNode(Node* node) {
// WARNING: potential dangling pointer loop under high throughput
node->prev->next = node->next;
node->next->prev = node->prev;
}
public:
LRUCache(int cap) : capacity(cap) {
head = new Node(-1, -1);
tail = new Node(-1, -1);
head->next = tail; tail->prev = head;
}
